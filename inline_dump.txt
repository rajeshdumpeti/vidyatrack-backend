   1 | # vidyatrack-backend
   2 | 
   3 | Backend service for the **VidyaTrack** platform.
   4 | 
   5 | This repository contains the backend API built using **FastAPI**, following an enterprise-grade, modular, and versioned architecture.  
   6 | The backend is designed to be **upgrade-friendly**, **maintainable**, and **production-ready**, starting with health checks and expanding toward authentication, RBAC, and database-backed features.
   7 | 
   8 | ---
   9 | 
  10 | ## Tech Stack
  11 | 
  12 | - **Language:** Python 3.12
  13 | - **Framework:** FastAPI
  14 | - **ASGI Server:** Uvicorn
  15 | - **Dependency Management:** pip + requirements.txt
  16 | - **(Upcoming):**
  17 |   - SQLAlchemy (ORM)
  18 |   - Alembic (DB migrations)
  19 |   - PostgreSQL
  20 | 
  21 | ---

   1 | 
   2 | from __future__ import annotations
   3 | 
   4 | from datetime import datetime, timezone, timedelta
   5 | from typing import Optional
   6 | 
   7 | from sqlalchemy.orm import Session
   8 | 
   9 | from app.db.models.notification_outbox import NotificationOutbox
  10 | 
  11 | 
  12 | def process_outbox_batch(
  13 |     db: Session,
  14 |     *,
  15 |     limit: int = 50,
  16 | ) -> int:
  17 |     """
  18 |     Phase 2 worker (synchronous, local-run only):
  19 |     - Fetch PENDING outbox rows
  20 |     - 'Send' by printing to console (no external integration)
  21 |     - Mark row SENT with sent_at
  22 |     - Retry-safe: if a row is already SENT, it will not be reprocessed
  23 |     """
  24 |     now = datetime.now(timezone.utc)
  25 |     rows = (
  26 |         db.query(NotificationOutbox)
  27 |         .filter(
  28 |             NotificationOutbox.status.in_(["PENDING", "FAILED"]),
  29 |             NotificationOutbox.next_attempt_at <= now,
  30 |         )
  31 |         .order_by(NotificationOutbox.id.asc())
  32 |         .limit(limit)
  33 |         .all())
  34 | 
  35 |     processed = 0
  36 |     MAX_ATTEMPTS = 5
  37 |     BACKOFF_MINUTES = [1, 5, 15, 60, 240]  # simple exponential-ish backoff
  38 | 
  39 |     for row in rows:
  40 |         try:
  41 |             # Simulated send (still stubbed)
  42 |             print(
  43 |                 f"[OUTBOX][SEND] id={row.id} event={row.event_type} "
  44 |                 f"to={row.recipient_phone} payload={row.payload}"
  45 |             )
  46 | 
  47 |             # raise Exception("test failure")
  48 | 
  49 |             # Success path
  50 |             row.status = "SENT"
  51 |             row.sent_at = now
  52 |             row.last_error = None
  53 | 
  54 |             processed += 1
  55 | 
  56 |         except Exception as exc:
  57 |             # Failure path
  58 |             row.attempts += 1
  59 |             row.status = "FAILED"
  60 |             row.last_error = str(exc)
  61 | 
  62 |             if row.attempts >= MAX_ATTEMPTS:
  63 |                 # Stop retrying
  64 |                 row.next_attempt_at = now + timedelta(days=365)
  65 |             else:
  66 |                 delay = BACKOFF_MINUTES[min(
  67 |                     row.attempts - 1, len(BACKOFF_MINUTES) - 1)]
  68 |                 row.next_attempt_at = now + timedelta(minutes=delay)
  69 | 
  70 |     if processed:
  71 |         db.commit()
  72 | 
  73 |     return processed
